📝 使用示例

// 模板中添加对话框组件
{{ render_three_choice_dialog(
    'unsavedChangesModal',
    '检测到未保存的数据更改', 
    '您有未保存的目标设置或合格值更改，请选择：',
    'fas fa-exclamation-triangle',
    [
        {'text': '保存', 'color': 'success', 'icon': 'fas fa-save', 'action': 'handleSaveAction'},
        {'text': '放弃', 'color': 'warning', 'icon': 'fas fa-trash', 'action': 'handleDiscardAction'},
        {'text': '取消', 'color': 'secondary', 'icon': 'fas fa-times', 'action': 'handleCancelAction'}
    ]
) }}

// JavaScript 中显示对话框
function showConfirmDialog() {
    const modal = new bootstrap.Modal(document.getElementById('unsavedChangesModal'));
    modal.show();
}
